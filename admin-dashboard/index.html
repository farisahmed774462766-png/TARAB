const http = require("http"); const { WebSocketServer } = require("ws"); const { TOTAL_SEATS, MIN_ENTRY_FEE, MAX_ENTRY_FEE, createTable, } = require("./table_manager"); const { createPlayerWallet, applyReward, settleDebtManually, } = require("./economy_manager"); const { joinTableWithWallet, leaveTable } = require("./game_session"); const { createPlayerRankProfile, recordMatchResult, resetSeasonPoints, createTournament, } = require("./season_manager"); const { createSocialProfile, sendFriendRequest, acceptFriendRequest, removeFriend, blockUser, createPrivateTableInvite, registerReferral, } = require("./social_manager"); const { createVipAccount, awardVipPoints, redeemVipPointsForCoins, evaluateSubscriptionStatus, } = require("./vip_manager"); const { startCardRound, playCard, } = require("./card_round_manager"); const { ensurePlayerAndWallet, saveWalletAndTransaction, saveTableState, } = require("./db"); const PORT = process.env.PORT || 3000; const DB_ENABLED = Boolean(process.env.DATABASE_URL); const tables = new Map(); const rounds = new Map(); const wallets = new Map(); const rankProfiles = new Map(); const socialProfiles = new Map(); const vipAccounts = new Map(); const tournaments = new Map(); async function getOrCreateWallet(playerId, displayName = null) { if (wallets.has(playerId)) { return wallets.get(playerId); } let wallet; if (DB_ENABLED) { try { wallet = await ensurePlayerAndWallet(playerId, displayName, 10000); } catch (_) { wallet = createPlayerWallet({ playerId, coins: 10000, debt: 0 }); } } else { wallet = createPlayerWallet({ playerId, coins: 10000, debt: 0 }); } wallets.set(playerId, wallet); return wallet; } function getOrCreateSocial(playerId) { if (!socialProfiles.has(playerId)) { socialProfiles.set(playerId, createSocialProfile({ playerId })); } return socialProfiles.get(playerId); } function getOrCreateVip(playerId) { if (!vipAccounts.has(playerId)) { vipAccounts.set(playerId, createVipAccount({ playerId })); } const account = vipAccounts.get(playerId); return evaluateSubscriptionStatus(account); } function getOrCreateRank(playerId, displayName = null) { if (!rankProfiles.has(playerId)) { rankProfiles.set( playerId, createPlayerRankProfile({ playerId, displayName: displayName || `لاعب ${playerId}`, }) ); } return rankProfiles.get(playerId); } async function persistWallet(wallet) { if (!DB_ENABLED) return; try { const lastTx = wallet.transactions[wallet.transactions.length - 1] || null; await saveWalletAndTransaction(wallet, lastTx); } catch (_) { // تجاهل خطأ الاتصال المؤقت بقاعدة البيانات حتى لا يتوقف الاتصال اللحظي } } async function persistTable(table) { if (!DB_ENABLED) return; try { await saveTableState(table); } catch (_) { // تجاهل خطأ الاتصال المؤقت بقاعدة البيانات } } const server = http.createServer((req, res) => { if (req.url === "/health") { res.writeHead(200, { "Content-Type": "application/json; charset=utf-8" }); res.end( JSON.stringify({ service: "TARAB Game Server", status: "running", dbPersistence: DB_ENABLED, seatsPerTable: TOTAL_SEATS, entryFeeRange: { min: MIN_ENTRY_FEE, max: MAX_ENTRY_FEE }, activeTables: tables.size, activeRounds: rounds.size, }) ); return; } res.writeHead(200, { "Content-Type": "text/plain; charset=utf-8" }); res.end("TARAB WebSocket & Game Server is active"); }); const wss = new WebSocketServer({ server }); wss.on("connection", (socket) => { socket.send( JSON.stringify({ type: "WELCOME", message: "مرحباً بك في خادم لعبة طراب (TARAB)", seatsPerTable: TOTAL_SEATS, }) ); socket.on("message", async (rawData) => { try { const event = JSON.parse(rawData.toString()); const payload = event.payload || {}; if (event.type === "CREATE_TABLE") { const tableId = payload.tableId || `TBL-${tables.size + 1}`; const table = createTable({ tableId, isPrivate: Boolean(payload.isPrivate), entryFee: Number(payload.entryFee ?? MIN_ENTRY_FEE), }); tables.set(tableId, table); await persistTable(table); socket.send(JSON.stringify({ type: "TABLE_CREATED", table })); return; } if (event.type === "JOIN_TABLE") { const table = tables.get(payload.tableId); if (!table) { socket.send(JSON.stringify({ type: "ERROR", message: "الطاولة غير موجودة" })); return; } const wallet = await getOrCreateWallet(payload.playerId, payload.playerName); const result = joinTableWithWallet({ table, wallet, playerName: payload.playerName, isBot: Boolean(payload.isBot), isConnected: true, }); if (result.joined) { await persistTable(result.table); if (!payload.isBot) { await persistWallet(result.wallet); } } socket.send(JSON.stringify({ type: "JOIN_RESULT", ...result })); return; } if (event.type === "LEAVE_TABLE") { const table = tables.get(payload.tableId); if (!table) { socket.send(JSON.stringify({ type: "ERROR", message: "الطاولة غير موجودة" })); return; } const result = leaveTable({ table, playerId: payload.playerId }); await persistTable(result.table); socket.send(JSON.stringify({ type: "LEAVE_RESULT", ...result })); return; } // أحداث جولة الأوراق if (event.type === "START_ROUND") { const table = tables.get(payload.tableId); if (!table) { socket.send(JSON.stringify({ type: "ERROR", message: "الطاولة غير موجودة" })); return; } const round = startCardRound(table, Number(payload.cardsPerPlayer ?? 5)); rounds.set(table.tableId, round); socket.send(JSON.stringify({ type: "ROUND_STARTED", round })); return; } if (event.type === "PLAY_CARD") { const tableId = payload.tableId; const round = rounds.get(tableId); if (!round) { socket.send(JSON.stringify({ type: "ERROR", message: "لا توجد جولة نشطة لهذه الطاولة" })); return; } const updatedRound = playCard(round, { playerId: payload.playerId, cardId: payload.cardId, }); if (updatedRound.status === "FINISHED" && updatedRound.winnerId) { const table = tables.get(tableId); if (table) { const winWallet = await getOrCreateWallet(updatedRound.winnerId); const potReward = table.entryFee * 2; applyReward(winWallet, potReward, "round_win"); await persistWallet(winWallet); } } socket.send(JSON.stringify({ type: "CARD_PLAYED", round: updatedRound })); return; } if (event.type === "APPLY_REWARD") { const wallet = await getOrCreateWallet(payload.playerId); applyReward(wallet, payload.amount, payload.source || "daily_reward"); await persistWallet(wallet); socket.send(JSON.stringify({ type: "WALLET_UPDATED", wallet })); return; } if (event.type === "SETTLE_DEBT") { const wallet = await getOrCreateWallet(payload.playerId); const settlement = settleDebtManually(wallet, payload.amount); await persistWallet(wallet); socket.send(JSON.stringify({ type: "DEBT_SETTLED", settlement, wallet })); return; } // أحداث الأصدقاء والدعوات والإحالات if (event.type === "SEND_FRIEND_REQUEST") { const sender = getOrCreateSocial(payload.senderId); const receiver = getOrCreateSocial(payload.receiverId); const resSocial = sendFriendRequest(sender, receiver); socket.send(JSON.stringify({ type: "FRIEND_REQUEST_SENT", ...resSocial })); return; } if (event.type === "ACCEPT_FRIEND_REQUEST") { const accepter = getOrCreateSocial(payload.accepterId); const requester = getOrCreateSocial(payload.requesterId); const resSocial = acceptFriendRequest(accepter, requester); socket.send(JSON.stringify({ type: "FRIEND_REQUEST_ACCEPTED", ...resSocial })); return; } if (event.type === "REMOVE_FRIEND") { const profile = getOrCreateSocial(payload.playerId); removeFriend(profile, payload.targetId); socket.send(JSON.stringify({ type: "FRIEND_REMOVED", profile })); return; } if (event.type === "BLOCK_USER") { const profile = getOrCreateSocial(payload.playerId); blockUser(profile, payload.targetId); socket.send(JSON.stringify({ type: "USER_BLOCKED", profile })); return; } if (event.type === "CREATE_PRIVATE_INVITE") { const invite = createPrivateTableInvite({ tableId: payload.tableId, hostId: payload.hostId, inviteCode: payload.inviteCode, }); socket.send(JSON.stringify({ type: "PRIVATE_INVITE_CREATED", invite })); return; } if (event.type === "REGISTER_REFERRAL") { const inviterSocial = getOrCreateSocial(payload.inviterId); const referral = registerReferral({ inviterProfile: inviterSocial, inviteeId: payload.inviteeId, referralRewardCoins: payload.rewardCoins ?? 500, }); const wallet = await getOrCreateWallet(payload.inviterId); applyReward(wallet, referral.rewardCoins, "friend_referral"); await persistWallet(wallet); socket.send(JSON.stringify({ type: "REFERRAL_REGISTERED", referral, wallet })); return; } // أحداث VIP ونقاط الولاء if (event.type === "AWARD_VIP_POINTS") { const account = getOrCreateVip(payload.playerId); const result = awardVipPoints(account, payload.points, payload.reason || "activity"); socket.send(JSON.stringify({ type: "VIP_POINTS_AWARDED", ...result })); return; } if (event.type === "REDEEM_VIP_POINTS") { const account = getOrCreateVip(payload.playerId); const redemption = redeemVipPointsForCoins(account, payload.pointsToRedeem); const wallet = await getOrCreateWallet(payload.playerId); applyReward(wallet, redemption.coinsEarned, "vip_points_redemption"); await persistWallet(wallet); socket.send(JSON.stringify({ type: "VIP_POINTS_REDEEMED", redemption, wallet })); return; } // أحداث المواسم والبطولات if (event.type === "RECORD_MATCH_RESULT") { const profile = getOrCreateRank(payload.playerId, payload.displayName); recordMatchResult(profile, { won: Boolean(payload.won), pointsDelta: payload.pointsDelta ?? 25, matchSummary: payload.matchSummary || "", }); socket.send(JSON.stringify({ type: "RANK_UPDATED", profile })); return; } if (event.type === "CREATE_TOURNAMENT") { const tournament = createTournament({ tournamentId: payload.tournamentId || `TRN-${tournaments.size + 1}`, title: payload.title || "بطولة طراب", cadence: payload.cadence || "DAILY", entryFee: payload.entryFee ?? 1000, tier: payload.tier || "STANDARD", }); tournaments.set(tournament.tournamentId, tournament); socket.send(JSON.stringify({ type: "TOURNAMENT_CREATED", tournament })); return; } if (event.type === "RESET_SEASON") { const updatedList = resetSeasonPoints( Array.from(rankProfiles.values()), payload.seasonLabel || "Season Reset" ); rankProfiles.clear(); for (const item of updatedList) { rankProfiles.set(item.playerId, item); } socket.send(JSON.stringify({ type: "SEASON_RESET", profiles: updatedList })); return; } socket.send( JSON.stringify({ type: "ACK", receivedType: event.type || "UNKNOWN", timestamp: new Date().toISOString(), }) ); } catch (err) { socket.send( JSON.stringify({ type: "ERROR", message: err.message || "صيغة الرسالة غير صالحة (JSON مطلوب)", }) ); } }); }); server.listen(PORT, () => { console.log(`TARAB server listening on port ${PORT}`); });
